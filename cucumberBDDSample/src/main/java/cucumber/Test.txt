        if (authContext != null && authContext.has("application/x-www-form-urlencoded")) {
            // 1. Get the raw string: "grant_type=client_credentials&client_id=...&client_secret=..."

            // 2. Extract client_id and client_secret from the string
            String clientId = extractParam(rawData, "client_id");
            String clientSecret = extractParam(rawData, "client_secret");

            if (clientId != null && clientSecret != null) {
                // 3. Create the Basic Auth Header
                String auth = clientId + ":" + clientSecret;
                String encodedAuth = Base64.getEncoder().encodeToString(auth.getBytes(StandardCharsets.UTF_8));
                conn.setRequestProperty("Authorization", "Basic " + encodedAuth);
            }
        }

        // ... (Existing Header and Body logic)
        if (headers != null) {
            for (Map.Entry<String, Object> entry : headers.entrySet()) {
                conn.setRequestProperty(entry.getKey(), entry.getValue().toString());
            }
        }
        
        // Finalize request...
        int status = conn.getResponseCode();
        
    } catch (Exception e) {
        e.printStackTrace();
    }
    return null;
}

/**
 * Helper method to extract specific values from the URL-encoded string
 */
private static String extractParam(String data, String param) {
    for (String pair : data.split("&")) {
        String[] parts = pair.split("=");
        if (parts.length == 2 && parts[0].equalsIgnoreCase(param)) {
            return parts[1];
        }
    }
    return null;
}
